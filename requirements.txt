# NCM Studio 的依赖
#
# 安装：
#   python3 -m venv .venv
#   .venv/bin/python -m pip install -r requirements.txt
#
# 或者用更快的 uv：
#   uv venv && uv pip install -r requirements.txt

# 界面框架。唯一必须的依赖，三个平台通用。
PyQt6>=6.5

# Windows 专用：伪终端（ConPTY）。
#
# 为什么非它不可：ncmctl download 一进来就调 pb.StartPool()，那个函数会去问
# 「我的标准输出是不是一块真终端」—— Windows 上走 GetConsoleMode，stdout 是
# 管道就直接失败退出，一个字节都下不了。而 Python 标准库在 Windows 上没有 pty，
# 想给子进程一块 ConPTY 就只能靠这个库。
#
# Linux / macOS 用标准库的 pty 模块，装了也没用，所以按平台条件安装。
# 打包时会连同它自带的 conpty.dll / OpenConsole.exe 一起进 exe，用户无感。
pywinpty>=2.0; sys_platform == "win32"

# 写入音频标签（下载页「歌词 → 写入音频标签」）。
#
# ncmctl 只下音频、不写任何标签（连 artist/title 都没有），所以「把歌词内嵌进
# 文件」这件事只能自己做。mutagen 认 flac / mp3 / m4a / ogg / opus 的标签格式，
# 是纯 Python 的单文件库（一个 wheel 196 KB，没有二进制依赖），PyInstaller 能
# 自动收全，spec 不用改。
#
# 只有「写入音频标签」这一种模式用得上：默认的「独立 .lrc 文件」自己拼字符串
# 就行。缺了它程序照常跑，只是选到内嵌模式时会给出明确提示。
# 许可 GPL-2.0-or-later，与本项目一致。
mutagen>=1.47

# Windows：SMTC 系统媒体控件（媒体键、任务栏「正在播放」与锁屏播放器）。
#
# 用 PyWinRT 的投影包，而不是把所有 Windows 命名空间打成一包的 winsdk：
# 每个命名空间单独分发，这里只装实际用到的几个，产物体积小得多。
# ncmgui/smtc.py 直接导入 winrt.runtime、Foundation、Media、Media.Playback
# 与 Storage.Streams；Media.Core 和 Storage 是播放对象签名里可能出现的类型，
# 一并装上，免得运行时按需导入失败后退化成「没有媒体控件」。
# 非 Windows 平台不装，Windows 上缺任何一个时整个功能静默关闭，不影响播放。
winrt-runtime~=3.2.1; sys_platform == "win32"
winrt-Windows.Foundation~=3.2.1; sys_platform == "win32"
winrt-Windows.Foundation.Collections~=3.2.1; sys_platform == "win32"
winrt-Windows.Media~=3.2.1; sys_platform == "win32"
winrt-Windows.Media.Core~=3.2.1; sys_platform == "win32"
winrt-Windows.Media.Playback~=3.2.1; sys_platform == "win32"
winrt-Windows.Storage~=3.2.1; sys_platform == "win32"
winrt-Windows.Storage.Streams~=3.2.1; sys_platform == "win32"

# Linux：MPRIS 媒体控制（蓝牙耳机 / 键盘上的播放暂停键）。
#
# Linux 桌面通过 MPRIS（D-Bus）把耳机的 AVRCP 指令转给播放器；mpv 自己不实现
# 这个接口，所以由 ncmgui/mpris.py 注册一份最小服务。dbus-next 是纯 Python
# 实现，不需要 libdbus；Windows/macOS 用的是各自系统的媒体控件，不装它。
dbus-next>=0.2.3; sys_platform == "linux"

# 自动更新的签名验证（Ed25519）。
#
# 安装包与 SHA256SUMS 来自同一个 Release，只比哈希没有独立信任根。发布者用
# 离线私钥给 SHA256SUMS 签名，本程序内置公钥验签，验不过就拒绝安装。
# 私钥不进 CI、不进仓库。PyInstaller 自带 cryptography 的 hook，无需额外配置。
# 缺失时程序照常运行，只是「设置 → 更新本程序」会拒绝自动安装并提示手动下载。
cryptography>=42

# 可选：二维码登录的兜底渲染。
# ncmctl 会自己生成 qrcode.png，本程序优先直接显示它；
# 只有在 PNG 没生成时，才用这个库把终端里打印的二维码内容渲染出来。
# 缺失时程序照常运行，只是在那种情况下给出文字提示。
qrcode[pil]>=7.4
